int index(int* arr, vector<int>& vec, int var){
	for (auto i : vec){
		if (arr[i] == var){
			return i;
		}
	}
	return -1;
}

int customer_index(vector<Customer> vec, Customer comp){
	for (int i = 0; i < vec.size(); i++){
		if (comp == vec[i]){
			return i;
		}
	}
	return -1;
}

vector<int> shortest_path(my_Graph& my_map, int source, int destination, int& sum_distance){

	const int total_locations = my_map.total_vertices;
	int start = source;

	int* arr = new int[total_locations];
	bool* visited = new bool[total_locations];
	int* prev = new int[total_locations];


	for (int i = 0; i < total_locations; i++){
		visited[i] = false;
		arr[i] = INT_MAX; //initialize to infinity
		prev[i] = -1;
	}

	arr[source] = 0;
	visited[source] = true;

	for (int i = 0; i < total_locations; i++){
		Min_Heap<int> f(total_locations);
		vector<int> vec;
		for (int j = 0; j < total_locations; j++){
			if (arr[j] == INT_MAX && my_map.matrix[start][j] != 0){
				arr[j] = my_map.matrix[start][j] + arr[start];
				prev[j] = start;
				f.insert(my_map.matrix[start][j] + arr[start]);
				vec.push_back(j);
			}
			else if (visited[j] != true && my_map.matrix[start][j] != 0){
				if (my_map.matrix[start][j] + arr[start] < arr[j]){
					arr[j] = my_map.matrix[start][j] + arr[start];
					prev[j] = start;
				}
				f.insert(arr[j]);
				vec.push_back(j);
			}
		}
		visited[start] = true;
		start = index(arr, vec, f.Delete());
		if (start == -1){
			break;
		}
	}

	vector<int> path;
	for (int at = destination; at != -1; at = prev[at]) {
		path.push_back(at);
	}
	reverse(path.begin(), path.end());

	int first = path[0];
	for (auto i : path){
		sum_distance = sum_distance + my_map.matrix[first][i];
		first = i;
	}

	delete[] prev;
	delete[] visited;
	delete[] arr;

	prev = NULL;
	visited = NULL;
	arr = NULL;

	return path;
}

void read_map(string file_name, my_Graph& new_map){
	ifstream read;

	read.open(file_name);

	int from;
	int to;
	int weight;

	while (!read.eof()){
		read >> from;
		read >> to;
		read >> weight;

		new_map.construct_road(from, to, weight);
	}

	read.close();
}
