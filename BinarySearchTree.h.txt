#ifndef BINARY_SEARCH_TREE_H
#define BINARY_SEARCH_TREE_H

#include "BNode.h"

#include <iostream>
#include <queue>

using namespace std;


template <class T>
class BinarySearchTree{

public:

	BinarySearchTree(void);

	bool insert(T);

	void levelOrder(void);

	bool search(T&, BNode<T>*&);

private:
	BNode<T>* root;
};

template <class T>
BinarySearchTree<T>::BinarySearchTree(void){
	root = NULL;
}

template <class T>
void BinarySearchTree<T>::levelOrder(void){
	if (root == NULL){
		return;
	}

	BNode<T>* temp = root;
	queue<BNode<T>*> q;

	q.emplace(root);

	while (q.empty() != true){
		temp = q.front();
		q.pop();

		cout << temp->getData() << " ";

		if (temp->getLeftChild()){
			q.emplace(temp->getLeftChild());
		}
		if (temp->getRightChild()){
			q.emplace(temp->getRightChild());
		}
	}
}

template <class T>
bool BinarySearchTree<T>::insert(T insert_val){

	BNode<T>* insert_node = new BNode<T>(insert_val);

	if (root == NULL){
		root = insert_node;
		return true;
	}

	BNode<T>* temp = root;
	BNode<T>* prev = NULL;

	while (temp != NULL){
		if (insert_val > temp->getData()){
			prev = temp;
			temp = temp->getRightChild();
		}
		else if (insert_val < temp->getData()){
			prev = temp;
			temp = temp->getLeftChild();
		}
	}

	if (insert_val > prev->getData()){
		prev->setRightChild(insert_node);
		return true;
	}
	else if (insert_val < prev->getData()){
		prev->setLeftChild(insert_node);
		return true;
	}
	return false;
}


template <class T>
bool BinarySearchTree<T>::search(T& val, BNode<T>*& rtn){
	
	BNode<T>* temp = root;

	while (temp != NULL){

		if (val == temp->getData()){
			rtn = temp;
			return true;
		}

		if (val > temp->getData()){
			temp = temp->getRightChild();
		}
		else if (val < temp->getData()){
			temp = temp->getLeftChild();
		}
	}

	if (temp == NULL){
		return false;
	}
}
